f_pinbarFlags(
     float candleOpen,
     float candleHigh,
     float candleLow,
     float candleClose,
     float bodyThreshold,
     float wickThreshold,
     float wickRatio) =>

    float candleRange = candleHigh - candleLow
    float candleBody = math.abs(candleClose - candleOpen)
    float upperWick = candleHigh - math.max(candleOpen, candleClose)
    float lowerWick = math.min(candleOpen, candleClose) - candleLow
    float bodyPct =
         candleRange > 0 ?
         candleBody / candleRange * 100 :
         100

    bool smallBody =
         candleRange > 0 and
         bodyPct <= bodyThreshold

    bool bullishPinbar =
         smallBody and
         lowerWick / candleRange * 100 >= wickThreshold and
         lowerWick >= upperWick * wickRatio

    bool bearishPinbar =
         smallBody and
         upperWick / candleRange * 100 >= wickThreshold and
         upperWick >= lowerWick * wickRatio

    [bullishPinbar or bearishPinbar, bullishPinbar, bearishPinbar]

f_normalizeCvdBase(string source) =>
    string result = source

    if na(result) or str.length(result) == 0
        result := syminfo.ticker

    if str.startswith(result, "100000")
        result := str.substring(result, 6, str.length(result))
    else if str.startswith(result, "10000")
        result := str.substring(result, 5, str.length(result))
    else if str.startswith(result, "1000")
        result := str.substring(result, 4, str.length(result))
    else if str.startswith(result, "100")
        result := str.substring(result, 3, str.length(result))
    else if str.startswith(result, "10")
        result := str.substring(result, 2, str.length(result))

    if str.endswith(result, "100000")
        result := str.substring(result, 0, str.length(result) - 6)
    else if str.endswith(result, "10000")
        result := str.substring(result, 0, str.length(result) - 5)
    else if str.endswith(result, "1000")
        result := str.substring(result, 0, str.length(result) - 4)
    else if str.endswith(result, "100")
        result := str.substring(result, 0, str.length(result) - 3)
    else if str.endswith(result, "10")
        result := str.substring(result, 0, str.length(result) - 2)

    result

// ---------- Process CVD ----------
f_processCvd(
     array<float> opens,
     array<float> closes,
     array<float> volumes,
     array<float> priorCloses,
     int previousDirection) =>

    float running = 0.0
    float aggressiveBuyVolume = 0.0
    float aggressiveSellVolume = 0.0
    float cvdRunHigh = 0.0
    float cvdRunLow = 0.0
    int count = 0
    int direction = previousDirection

    if not na(closes)
        count := array.size(closes)

        if count > 0
            for i = 0 to count - 1
                float intrabarOpen = array.get(opens, i)
                float intrabarClose = array.get(closes, i)
                float intrabarVolume = nz(array.get(volumes, i), 0.0)
                float previousClose = na

                if i == 0
                    previousClose := array.get(priorCloses, i)
                else
                    previousClose := array.get(closes, i - 1)

                // Candle direction first, then close-to-close for flat candles.
                if intrabarClose > intrabarOpen
                    direction := 1
                else if intrabarClose < intrabarOpen
                    direction := -1
                else if not na(previousClose) and intrabarClose > previousClose
                    direction := 1
                else if not na(previousClose) and intrabarClose < previousClose
                    direction := -1

                running += direction * intrabarVolume

                if direction > 0
                    aggressiveBuyVolume += intrabarVolume
                else if direction < 0
                    aggressiveSellVolume += intrabarVolume

                cvdRunHigh := math.max(cvdRunHigh, running)
                cvdRunLow := math.min(cvdRunLow, running)

    [running, aggressiveBuyVolume, aggressiveSellVolume, cvdRunHigh, cvdRunLow, count, direction]

    cvdLow := na
    cvdClose := na

// ---------- Candle-shape measurements ----------
// OI uses the same candle that is plotted below. The body is compared with the
// complete high-low range so a small body with a large wick is treated as flat.
oiOpen = nz(totalC[1], totalC)
oiClose = totalC
oiHigh = math.max(oiOpen, oiClose, totalH)
oiLow = math.min(oiOpen, oiClose, totalL)

oiRange = oiHigh - oiLow
oiBody = math.abs(oiClose - oiOpen)
oiBodyPct =
     oiRange > 0 ?
     oiBody / oiRange * 100 :
     0

// Low-activity checks compare absolute changes, so positive and negative
// candles do not cancel each other in the moving average.
oiActivity = math.abs(oi_delta)
oiActivityAvg = ta.sma(oiActivity, flatActivityLength)
oi_is_low_activity =
     not na(oiActivityAvg) and
     oiActivityAvg > 0 and
     oiActivity <= oiActivityAvg / flatActivityFactor

oi_is_flat =
     oiRange == 0 or
     oiBodyPct <= flatBodyPct or
     oi_is_low_activity

// Pinbar detection is intentionally stricter than flat detection. A pinbar
// is still flat by the 25% rule, but it additionally needs one dominant wick.
[oi_is_pinbar, oi_is_bullish_pinbar, oi_is_bearish_pinbar] =
     f_pinbarFlags(
         oiOpen,
         oiHigh,
         oiLow,
         oiClose,
         pinbarBodyPct,
         pinbarWickPct,
         pinbarWickRatio)

cvdRange =
     not na(cvdHigh) and not na(cvdLow) ?
     cvdHigh - cvdLow :
     na

cvdBody =
     not na(cvdClose) and not na(cvdOpen) ?
     math.abs(cvdClose - cvdOpen) :
     na

cvdBodyPct =
     not na(cvdRange) and cvdRange > 0 ?
     cvdBody / cvdRange * 100 :
     0

cvdActivity = not na(vol_delta) ? math.abs(vol_delta) : na
cvdActivityAvg = ta.sma(cvdActivity, flatActivityLength)
cvd_is_low_activity =
     not na(cvdActivity) and
     not na(cvdActivityAvg) and
     cvdActivityAvg > 0 and
     cvdActivity <= cvdActivityAvg / flatActivityFactor

     hasCvdIntrabars and
     (cvdRange == 0 or cvdBodyPct <= flatBodyPct or cvd_is_low_activity)

         cvdOpen,
         cvdHigh,

         pinbarWickPct,

var bool inSpikeSeq = false
var float initial_open_oi = na

// A sequence has its own peak. This is deliberately independent of the
// TradingView alert object: creating an alert must never establish the peak.
var float cumulative_high_oi = na
var int cumulative_high_bar_index = na
var int bars_since_new_high = 0
var label active_lbl = na

// These flags are only per-peak guards. They are re-armed when OI makes a
// higher high, so pullback percentages are always measured from that high.
var bool alert1_fired = false
var bool alert2_fired = false

bool trigger_alert1 = false
bool trigger_alert2 = false

is_single_spike =
     not inSpikeSeq and
     oi_body_delta > 0 and
     oi_body_delta >= spikeMult * nz(avg_oi_delta[1], avg_oi_delta)

is_10_green_spike =
     not inSpikeSeq and
     green_streak == 10

is_initial_spike =
     is_single_spike or is_10_green_spike

if is_initial_spike
    inSpikeSeq := true
    initial_open_oi := is_10_green_spike ? o[9] : o

    cumulative_high_oi := h
    cumulative_high_bar_index := bar_index

    bars_since_new_high := 0
    alert1_fired := false
    alert2_fired := false

    peak_cum_change = cumulative_high_oi - initial_open_oi
    peak_pct_change =
         initial_open_oi > 0 ?
         peak_cum_change / initial_open_oi * 100 :
         0

    lbl_text = str.tostring(peak_pct_change, "#.##") + "%"
    lbl_color =
         peak_pct_change > 30.0 ?
         color.new(#b91c1c, 0) :
         color.new(#15803d, 0)

    lbl_y = cumulative_high_oi + avg_oi_delta * 0.3

         yloc=yloc.price,
         size=size.normal)

    // candle's high therefore becomes the reference immediately, including
    // when the same candle later closes lower.

    if made_new_sequence_high
        cumulative_high_oi := h
        cumulative_high_bar_index := bar_index
        bars_since_new_high := 0

        // A new peak starts a new pullback measurement.
        alert1_fired := false
        alert2_fired := false
    else
        bars_since_new_high += 1

    peak_cum_change = cumulative_high_oi - initial_open_oi
    peak_pct_change =
         initial_open_oi > 0 ?
         peak_cum_change / initial_open_oi * 100 :
         0

         color.new(#b91c1c, 0) :
         color.new(#15803d, 0)

    lbl_y = cumulative_high_oi + avg_oi_delta * 0.3

    label.set_xy(active_lbl, cumulative_high_bar_index, lbl_y)
    label.set_text(active_lbl, lbl_text)
    label.set_color(active_lbl, lbl_color)

if inSpikeSeq
    // The pullback is a retracement of the active OI expansion, not a
    // percentage of the absolute OI value.
    //
    // Example:
    //   qualifying OI = 510M
    //   highest OI    = 592M
    //   active spike  = 82M
    //   10% pullback  = 8.2M
    //
    // Therefore the 10% threshold is 583.8M, not 532.8M.
    spike_range =
         cumulative_high_oi - initial_open_oi

    pullback_amount =
         cumulative_high_oi - c

    pullback_ratio =
         spike_range > 0 ?
         pullback_amount / spike_range * 100 :
         0

    // Always use the sequence's highest OI, never the OI at the moment
    // the TradingView alert was created.
    if pullback_ratio >= pbAlert1 and not alert1_fired
        alert1_fired := true
        trigger_alert1 := true

    if pullback_ratio >= pbAlert2 and not alert2_fired
        alert2_fired := true
        trigger_alert2 := true

    // Expire the sequence after alerts are evaluated on this bar. This
    // prevents the reset candle from suppressing a valid pullback alert.
    sequence_expired =
         bars_since_new_high >= maxSidewaysBars or
         c < initial_open_oi

    if sequence_expired
        inSpikeSeq := false
        cumulative_high_oi := na
        cumulative_high_bar_index := na
        bars_since_new_high := 0
        alert1_fired := false
        alert2_fired := false

// ==============================================================================
// === ALERTS
// ==============================================================================

alertcondition(
     trigger_alert1,
     title="10% Pullback Alert",
     message="OI Pullback Triggered! (10% retracement of active OI expansion)")

alertcondition(
     trigger_alert2,
     title="20% Pullback Alert",
     message="OI Pullback Triggered! (20% retracement of active OI expansion)")
